local function um(jd, je)
    local kr = workspace:FindFirstChild("World")
    if not kr then
        return false
    end
    kr = kr:FindFirstChild("FX")
    if not kr then
        return false
    end
    if not kr:FindFirstChild("ViewModel") then
        return false
    end
    if jd == 0 and je == 0 then
        return true
    end
    local yk = workspace.CurrentCamera.ViewportSize
    local gd, ge = yk.X / 2, yk.Y / 2
    if np() then
        pcall(function()
            cx:SendTouchEvent(0, 0, gd, ge)
            task.wait()
            cx:SendTouchEvent(0, 1, gd + jd * 2, ge + je * 2)
            task.wait()
            cx:SendTouchEvent(0, 2, gd + jd * 2, ge + je * 2)
        end)
        return true
    else
        if ns() and type(mousemoverel) == "function" then
            mousemoverel(jd, je)
            return true
        end
    end
    return false
end
local function ul(mh)
    local qr = tick()
    local kr = workspace:FindFirstChild("World")
    if not kr then
        return false
    end
    kr = kr:FindFirstChild("FX")
    if not kr then
        return false
    end
    if not kr:FindFirstChild("ViewModel") then
        return false
    end
    if np() then
        pcall(function()
            local yk = workspace.CurrentCamera.ViewportSize
            local gc = Vector2.new(yk.X / 2, yk.Y / 2)
            cx:SendTouchEvent(0, 0, gc.X, gc.Y)
            if mh and mh > 0 then
                task.wait(mh)
            else
                task.wait(0.05)
            end
            cx:SendTouchEvent(0, 2, gc.X, gc.Y)
        end)
        return true
    else
        if not ns() then
            return false
        end
        if type(mouse1press) == "function" and type(mouse1release) == "function" then
            mouse1press()
            if mh and mh > 0 then
                task.wait(mh)
            end
            mouse1release()
            return true
        end
        if type(mouse1click) == "function" then
            mouse1click()
            return true
        end
    end
    return false
end

local function cg(gb, yo, vs, ng)
    local wk = gb .. "." .. yo
    local hp = n[wk] or "base"
    if hp == vs then
        return 
    end
    n[wk] = vs
    if not ng and Options and Options.SavedSkinsData then
        pcall(function()
            Options.SavedSkinsData:SetValue(ay:JSONEncode(n))
        end)
    end
    task.spawn(function()
        local ek = bs:WaitForChild("Assets", 10)
        if not ek then
            return 
        end
        local oe = ek:WaitForChild("Items", 10)
        if not oe then
            return 
        end
        local ga = oe:WaitForChild(gb, 10)
        if not ga then
            return 
        end
        local yq = ga:WaitForChild(yo, 10)
        if not yq then
            return 
        end
        local eq = yq:WaitForChild("base", 10)
        if not eq then
            return 
        end
        if hp ~= "base" then
            local hs = yq:FindFirstChild(hp)
            if hs and eq then
                local dm = eq:FindFirstChild("model")
                local dl = eq:FindFirstChild("animations")
                local tm = hs:FindFirstChild("model")
                local tl = hs:FindFirstChild("animations")
                if dm then
                    dm.Parent = hs
                end
                if dl then
                    dl.Parent = hs
                end
                if tm then
                    tm.Parent = eq
                end
                if tl then
                    tl.Parent = eq
                end
            end
        end
        if vs ~= "base" then
            local vo = yq:WaitForChild(vs, 10)
            if vo and eq then
                local wt = vo:FindFirstChild("model")
                local ws = vo:FindFirstChild("animations")
                local er = eq:FindFirstChild("model")
                local eo = eq:FindFirstChild("animations")
                if wt then
                    wt.Parent = eq
                end
                if ws then
                    ws.Parent = eq
                end
                if er then
                    er.Parent = vo
                end
                if eo then
                    eo.Parent = vo
                end
            end
        end
    end)
end

local function ja(rr, rs)
    if not ab.HitTracersEnabled then
        return 
    end
    if typeof(rr) ~= 'Vector3' or typeof(rs) ~= 'Vector3' then
        return 
    end
    local ip = rs - rr
    local ou = ip.Magnitude
    if ou <= 0.05 then
        return 
    end
    local xc = math.clamp(ab.HitTracerThickness or 0.08, 0.01, 1)
    local js = math.max(ab.HitTracerFadeIn or 0, 0)
    local jt = math.max(ab.HitTracerFadeOut or 0, 0)
    local ov = math.max(ab.HitTracerLifetime or 0.75, js + jt + 0.01)
    local gx = ab.HitTracerColor or Color3.fromRGB(0, 255, 255)
    local xk = Instance.new('Part')
    xk.Name = 'NeonHitTracer'
    xk.Anchored = true
    xk.CanCollide = false
    xk.CanTouch = false
    xk.CanQuery = false
    xk.CastShadow = false
    xk.Material = Enum.Material.Neon
    xk.Color = gx
    xk.Transparency = 1
    xk.Size = Vector3.new(xc, xc, ou)
    xk.CFrame = CFrame.lookAt((rr + rs) * 0.5, rs)
    xk.Parent = av
    e[xk] = true
    if js > 0 then
        cl:Create(xk, TweenInfo.new(js, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Transparency = 0,
        }):Play()
    else
        xk.Transparency = 0
    end
    task.delay(math.max(ov - jt, js), function()
        if not xk or not xk.Parent then
            return 
        end
        local xq = cl:Create(xk, TweenInfo.new(jt, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
            Transparency = 1,
        })
        xq:Play()
        xq.Completed:Connect(function()
            gr(xk)
        end)
        if jt <= 0 then
            gr(xk)
        end
    end)
end

local function ku(sd)
    local pz, qa, qb = mn, mn, mn
    local pt, pu, pv = -mn, -mn, -mn
    for mp = 1, #sd do
        local rq = sd[mp]
        local gf, wn = rq.CFrame, rq.Size
        local wl, wm, wo = wn.X * 0.5, wn.Y * 0.5, wn.Z * 0.5
        local hk = {
            gf * Vector3.new(wl, wm, wo),
            gf * Vector3.new(-wl, wm, wo),
            gf * Vector3.new(wl, -wm, wo),
            gf * Vector3.new(-wl, -wm, wo),
            gf * Vector3.new(wl, wm, -wo),
            gf * Vector3.new(-wl, wm, -wo),
            gf * Vector3.new(wl, -wm, -wo),
            gf * Vector3.new(-wl, -wm, -wo),
        }
        for of = 1, 8 do
            local xy = hk[of]
            if xy.X < pz then
                pz = xy.X
            end
            if xy.Y < qa then
                qa = xy.Y
            end
            if xy.Z < qb then
                qb = xy.Z
            end
            if xy.X > pt then
                pt = xy.X
            end
            if xy.Y > pu then
                pu = xy.Y
            end
            if xy.Z > pv then
                pv = xy.Z
            end
        end
    end
    local gc = Vector3.new((pz + pt) * 0.5, (qa + pu) * 0.5, (qb + pv) * 0.5)
    local vi = Vector3.new(pt - pz, pu - qa, pv - qb)
    return CFrame.new(gc), vi
end

local function cg_WatchEntities()

        cg_EntitiesConn:Disconnect()
        cg_EntitiesConn = nil
    end
    if cg_EntitiesChildConn then
        cg_EntitiesChildConn:Disconnect()
        cg_EntitiesChildConn = nil
    end
    local yt = workspace:FindFirstChild("World")
    local en = yt and yt:FindFirstChild("Entities")
    if not en then
        -- Wait for it
        task.spawn(function()
            local yt2 = workspace:WaitForChild("World", 30)
            local en2 = yt2 and yt2:WaitForChild("Entities", 30)
            if en2 and ab.AllowDummys then
                cg_ScanEntities()
                cg_EntitiesChildConn = en2.ChildAdded:Connect(function(go)
                    if ab.AllowDummys then cg_AddEntity(go) end
                end)
                cg_EntitiesConn = en2.ChildRemoved:Connect(function(go)
                    cg_RemoveEntity(go)
                end)
            end
        end)
        return
    end
    if ab.AllowDummys then cg_ScanEntities() end
    cg_EntitiesChildConn = en.ChildAdded:Connect(function(go)
        if ab.AllowDummys then cg_AddEntity(go) end
    end)
    cg_EntitiesConn = en.ChildRemoved:Connect(function(go)

    local xi = ht and ht.player or lg(qe)
    if xi == bg then
        return true
    end
    if not ab.TeamCheck then
        return false
    end
    if lx(qe) then
        return true
    end
    if xi and bg.Team ~= nil and xi.Team ~= nil then
        return bg.Team == xi.Team
    end
    return false
end
local function nw(wu, qe, pe, iw)
    if not wu or not wu:IsA("BasePart") then
        return false
    end
    local qr = tick()
    if qe then
        local fg = dh[qe]
        if fg and qr - fg.t < (ab.RaycastUpdateRate / 1000) then
            return fg.v
        end
    end
    local tj = u.CFrame.Position
    local it = wu.Position - tj
    local rz = RaycastParams.new()
    rz.FilterType = Enum.RaycastFilterType.Exclude
    local kd = {
    }
    if pe then
        table.insert(kd, pe)
    end
    if iw then
        for pm in pairs(jb) do
            if pm ~= pe then
                local uf = kw(pm, "HumanoidRootPart") or pm.PrimaryPart or kw(pm, "Head")
                if uf and (iw - uf.Position).Magnitude <= 5 then
                    table.insert(kd, pm)
                end
            end
        end
    end
    rz.FilterDescendantsInstances = kd
    local ue = workspace:Raycast(tj, it, rz)
    local xy = (not ue) or ue.Instance:IsDescendantOf(wu.Parent)
    if qe then
        dh[qe] = {
            v = xy,
            t = qr,
        }
    end
    return xy
end

local function lh(qe, ex)
    local sb = qe:FindFirstChild(ex)
    if not sb then
        return nil
    end
    local sw = sb.Position
    local qr = tick()
    local ta = sw
    if ab.AimbotPrediction then
        local pk = op[qe]
        local pl = ot[qe]
        if pk and pl then
            local jc = qr - pl
            if jc > 0 then
                local ye = (sw - pk) / jc
                ta = sw + ye * ab.AimbotPredictScale
            end
        end
    end
    op[qe] = sw
    ot[qe] = qr
    return ta
end

local function kx(pe, ri)
    local ev, eu = nil, mn
    local et = "Head"
    local yk = u.ViewportSize
    local gc = Vector2.new(yk.X / 2, yk.Y / 2)
    for qe in pairs(jb) do
        if qe == pe then
            continue
        end
        if qe.Name:sub(1, 7) == "Dummy_" and not ab.AllowDummys then
            continue
        end
        if qe:FindFirstChild("FinisherOverheadTemplate") then
            continue
        end
        if nj(qe) then
            continue
        end
        local ly = kw(qe, "Head")
        if not ly then
            continue
        end
        if ly:IsA("BasePart") and ly.Transparency == 1 then
            continue
        end
        if ab.AimbotWallcheck and not nw(ly, qe, pe, ri) then
            continue
        end
        local es = mn
        local hq = nil
        local lw = false
        if type(ab.AimbotBone) == "table" then
            for en, dk in pairs(ab.AimbotBone) do
                if dk then
                    lw = true
                    local ey = type(dk) == "string" and dk or en
                    local rq = kw(qe, ey)
                    if rq then
                        local wa, re = u:WorldToViewportPoint(rq.Position)
                        if re and wa.Z > 0 then
                            local ht = (Vector2.new(wa.X, wa.Y) - gc).Magnitude
                            if ht < es then
                                es = ht
                                hq = ey
                            end
                        end
                    end
                end
            end
        end
        if not lw then
            local wa, re = u:WorldToViewportPoint(ly.Position)
            if re and wa.Z > 0 then
                es = (Vector2.new(wa.X, wa.Y) - gc).Magnitude
                hq = "Head"
            end
        end
        if not hq then
            continue
        end
        local uf = kw(qe, "HumanoidRootPart") or qe.PrimaryPart or ly
        local iv = (ri - uf.Position).Magnitude
        if iv <= 5 or iv > ab.AimbotMaxDistance then
            continue
        end
        if es > ab.AimbotFOV then
            continue
        end
        if es < eu then
            eu = es
            ev = qe
            et = hq
        end
    end
    return ev, et
end

local function lf(sb)
    if not sb or not sb:IsA("BasePart") then
        return 0
    end
    local iv = (u.CFrame.Position - sb.Position).Magnitude
    if iv <= 0 then
        return 0
    end
    local vi = sb.Size
    local py = math.min(vi.X, vi.Y, vi.Z)
    local yu = py / 2
    local ko = math.rad(u.FieldOfView)
    local uu = (yu * u.ViewportSize.Y) / (2 * iv * math.tan(ko / 2))
    return uu
end

local function li()
    local yc = gq(ab.AimbotSensitivityify or 1, 0.1, 2)
    if yc <= 1 then
        return 0.5 + ((yc - 0.1) / 0.9) * 0.5
    end
    return yc
end
local function ds(qe, ey)
    if not qe then
        return 
    end
    local sb = kw(qe, ey) or kw(qe, "Head")
    if not sb then
        return 
    end
    local sw = lh(qe, ey) or sb.Position
    local wa, re = u:WorldToViewportPoint(sw)
    if not re then
        return 
    end
    local qi = co:GetMouseLocation()
    local jd = wa.X - qi.X
    local je = wa.Y - qi.Y
    local vx = gq(ab.AimbotSmoothing, 1, 100)
    local jr = (1 / vx) * li()
    if dj(jd) <= 1 and dj(je) <= 1 then
        um(jd * li(), je * li())
        return 
    end
    um(jd * jr, je * jr)
end

local function nk(wr)

    end

    end

    if not kr then
        return false
    end
    local ww = kw(wr, "HumanoidRootPart") or wr.PrimaryPart or kw(wr, "Head")
    if not ww then
        return false
    end
    local wv = ww.Position
    local gv, gw = nil, math.huge
    for _, yi in ipairs(kr:GetChildren()) do
        local mi = yi:FindFirstChild("HumanoidRootPart")
        if mi and mi:IsA("BasePart") then
            local iv = (mi.Position - wv).Magnitude
            if iv < gw then
                gv = yi
                gw = iv
            end
        end
    end
    if not gv then
        return false
    end
    local qf = gv:FindFirstChild("model")
    if qf then
        local lu = qf:FindFirstChild("Handle")
        if lu and lu:FindFirstChild("glare_attachment") then
            return true
        end
    end

local function xn()
    if dd then
        return 
    end

        local nt = false

            local hm = ss:FindFirstChild("Crosshair")
            if hm then
                local ut = hm:FindFirstChild("Scope")
                if ut and ut.Visible then
                    nt = true
                end
            end
        end
        if not nt then
            return 
        end
    end
    local qr = tick()
    if (qr - de) < df then
        return 
    end
    dd = true
    task.spawn(function()
        ul(ab.TriggerbotClickHold / 1000)
        de = tick()
        local pb = math.min(ab.TriggerbotMinDelay, ab.TriggerbotMaxDelay)
        local mc = math.max(ab.TriggerbotMinDelay, ab.TriggerbotMaxDelay)
        df = (pb + math.random() * (mc - pb)) / 1000
        dd = false
    end)
end

                            ColorSequenceKeypoint.new(0, jl),
                            ColorSequenceKeypoint.new(1, jl),
                        })
                        ox.gradient.Rotation = 0
                    end
                end
            else
                ht.guiBox.box.Visible = false
                ht.guiBox.corner.Visible = false
            end
        end
        if ab.ESPDistance then
            ht.distTxt.Color = iy
            ht.distTxt.Text = kl(iv / 3) .. "m"
            ht.distTxt.Position = Vector2.new(xe.X + wn.X / 2, fc.Y + 2)
            ht.distTxt.Visible = true
        else
            ht.distTxt.Visible = false
        end
        if ab.ESPSkeleton then
            local hr = vk
            if ab.ESPAnimMode == 'Spinning Gradient' then
                hr = ep:Lerp(jm, te)
            end
            for mp, ry in ipairs(bw) do
                local rs = kw(qe, ry[1])
                local rt = kw(qe, ry[2])
                local ow = ht.skeleton[mp]
                if rs and rt then
                    local uj, rf = yv(rs.Position)
                    local uk, rg = yv(rt.Position)
                    if rf and rg then
                        ow.From = uj
                        ow.To = uk
                        ow.Color = hr
                        ow.Thickness = ab.ESPSkeletonThickness or 1
                        ow.Visible = true
                    else
                        ow.Visible = false
                    end
                else
                    ow.Visible = false
                end
            end
        else
            for _, ow in ipairs(ht.skeleton) do
                ow.Visible = false
            end
        end
    end
end)
